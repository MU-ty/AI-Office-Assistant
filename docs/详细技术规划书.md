# 关键句提取算法 - TextRank
class TextRankExtractor:
    def __init__(self, top_k: int = 5):
        self.top_k = top_k
        self.graph = {}
    
    def extract_sentences(self, text: str) -> List[str]:
        # 1. 分句
        sentences = self._segment_sentences(text)
        
        # 2. 构建句子相似度图
        similarity_matrix = self._build_similarity_matrix(sentences)
        
        # 3. PageRank计算重要度
        scores = self._pagerank(similarity_matrix)
        
        # 4. 提取Top-K句子
        top_indices = np.argsort(scores)[-self.top_k:]
        return [sentences[i] for i in sorted(top_indices)]

# Action Items提取算法 - 动词-宾语依存分析
class ActionItemExtractor:
    def extract_actions(self, text: str) -> List[Dict]:
        doc = nlp(text)
        action_items = []
        
        for sent in doc.sents:
            # 找出VERB + NOUN模式
            for token in sent:
                if token.pos_ == "VERB":
                    # 提取动词和其宾语
                    verb = token.text
                    obj = self._get_verb_object(token)
                    actor = self._identify_actor(sent, token)
                    
                    action_items.append({
                        'action': verb,
                        'object': obj,
                        'actor': actor,
                        'sentence': sent.text
                    })
        
        return action_items

class HierarchicalSummarizer:
    def __init__(self, model_name: str = "facebook/bart-large-cnn"):
        self.summarizer = pipeline("summarization", model=model_name)

# 关键概念提取 - 学术术语识别
class AcademicConceptExtractor:
    def __init__(self):
        self.nlp = spacy.load("en_core_sci_sm")
        self.concept_db = self._load_academic_terms()
    
    def extract_concepts(self, text: str) -> List[Dict]:
        doc = self.nlp(text)
        concepts = []
        
        # 1. NER识别
        for ent in doc.ents:
            if ent.label_ in ["ORG", "PRODUCT", "GPE"]:
                concepts.append({
                    'term': ent.text,
                    'type': ent.label_,
                    'position': ent.start_char,
                    'confidence': 0.95
                })
        
        # 2. 学术术语匹配
        for token in doc:
            if token.text.lower() in self.concept_db:
                concepts.append({
                    'term': token.text,
                    'type': 'ACADEMIC_TERM',
                    'definition': self.concept_db[token.text.lower()],
                    'frequency': text.lower().count(token.text.lower())
                })
        
        return sorted(concepts, key=lambda x: x.get('frequency', 0), reverse=True)

from pinecone import Pinecone, ServerlessSpec

class PineconeVectorStore:

                name=self.index_name,
                dimension=384,  # Sentence-BERT输出维度
                metric="cosine",
                spec=ServerlessSpec(
                    cloud="aws",
                    region="us-east-1"
                )
            )
    
    def upsert_documents(self, documents: List[Dict]):
        """向量化并上传文档"""
        index = self.pc.Index(self.index_name)
        
        vectors_to_upsert = []
        for doc in documents:
            # 生成向量
            embedding = self._generate_embedding(doc['text'])
            
            vectors_to_upsert.append({
                'id': doc['id'],
                'values': embedding,
                'metadata': {
                    'doc_id': doc['document_id'],
                    'title': doc['title'],
                    'chunk_index': doc['chunk_index']
                }
            })
        
        index.upsert(vectors=vectors_to_upsert)
    
    def search_similar(self, query: str, top_k: int = 5) -> List[Dict]:
        """搜索相似文献"""
        index = self.pc.Index(self.index_name)
        query_embedding = self._generate_embedding(query)
        
        results = index.query(
            vector=query_embedding,
            top_k=top_k,
            include_metadata=True
        )
        
        return results['matches']

    def __init__(self):

        issues.extend(style_issues)
        
        # 4. 逻辑连贯性分析
        coherence_analysis = self._analyze_coherence(text)
        
        # 5. 可读性评分
        readability = self.readability_calculator.calculate(text)
        
        # 6. 重复率检测
        uniqueness_score = self._calculate_uniqueness(text)

            'grammar_issues': grammar_issues,
            'style_suggestions': term_suggestions,
            'coherence_score': coherence_analysis['score'],
            'readability_score': readability,
            'uniqueness_score': uniqueness_score,
            'overall_quality': self._calculate_overall_quality(
                grammar_issues, term_suggestions, coherence_analysis
            ),
            'recommendations': self._generate_recommendations(
                grammar_issues, coherence_analysis, readability
            )
        }
    
    def _check_grammar(self, text: str) -> List[Dict]:
        """语法检查"""
        matches = self.tool.check(text)
        issues = []
        
        for match in matches:
            if match.category not in ['WHITESPACE']:  # 忽略空白问题
                issues.append({
                    'type': 'grammar',
                    'position': match.offset,
                    'length': match.length,
                    'message': match.message,
                    'replacement': match.replacements[0] if match.replacements else None,
                    'severity': self._calculate_severity(match.ruleId)
                })
        
        return issues
    
    def _suggest_academic_terms(self, text: str) -> List[Dict]:
        """学术术语建议"""
        doc = self.nlp(text)
        suggestions = []
        
        for token in doc:
            if token.pos_ in ['NOUN', 'VERB', 'ADJ']:
                # 在学术术语库中查找替代词
                term_lower = token.text.lower()
                if term_lower in self.academic_terms:
                    suggestions.append({
                        'original': token.text,
                        'type': 'academic_term',
                        'position': token.idx,
                        'suggestions': self.academic_terms[term_lower],
                        'explanation': f"Consider using more academic language"
                    })
        
        return suggestions

        elif score >= 70:
            return "Fairly Easy"
        elif score >= 60:
            return "Standard"
        elif score >= 50:
            return "Fairly Difficult"
        elif score >= 30:
            return "Difficult"

            return "Very Difficult"

from langdetect import detect, detect_langs
import jieba
from transformers import pipeline

class MultilingualProcessor:
    def __init__(self):
        self.translator = translate_v2.Client()
        self.terminology_db = self._load_terminology_database()
        self.nlp_models = self._load_nlp_models()
        self.quality_evaluator = TranslationQualityEvaluator()
    
    SUPPORTED_LANGUAGES = {
        'zh': 'Chinese',
        'en': 'English',
        'ja': 'Japanese',
        'ko': 'Korean',
        'fr': 'French',
        'de': 'German',
        'es': 'Spanish',
        'ru': 'Russian'
    }
    
    def detect_language(self, text: str) -> Dict[str, any]:
        """检测文本语言"""
        try:
            language = detect(text)
            probabilities = detect_langs(text)
            
            return {
                'primary_language': language,
                'language_name': self.SUPPORTED_LANGUAGES.get(language, 'Unknown'),
                'probabilities': [(lang.lang, lang.prob) for lang in probabilities],
                'confidence': max([p.prob for p in probabilities])
            }
        except Exception as e:
            return {'error': str(e)}
    
    def translate_with_terminology(self, text: str, source_lang: str, 
                                  target_lang: str, domain: str = None) -> Dict:
        """使用术语库的翻译"""
        # 1. 分句处理
        sentences = self._segment_sentences(text, source_lang)
        
        # 2. 术语提取与替换
        terminology_map = {}
        processed_text = text
        
        for term in self._extract_terminology(text, source_lang, domain):
            if term['original'] in self.terminology_db:
                target_term = self.terminology_db[term['original']].get(
                    target_lang, term['original']
                )
                terminology_map[term['original']] = target_term
        
        # 3. 翻译
        translation_result = self.translator.translate_text(
            content=text,
            source_language_code=source_lang,
            target_language_code=target_lang
        )
        
        translated_text = translation_result['translatedText']
        
        # 4. 术语替换
        for original, target in terminology_map.items():
            # 使用更聪明的替换逻辑避免误替
            translated_text = self._smart_replace(
                translated_text, self._translate_term(original, source_lang, target_lang),
                target
            )
        
        # 5. 质量评估
        quality_score = self.quality_evaluator.evaluate(
            original_text=text,
            translated_text=translated_text,
            source_lang=source_lang,
            target_lang=target_lang
        )
        
        return {
            'original_text': text,
            'translated_text': translated_text,
            'source_language': source_lang,
            'target_language': target_lang,
            'terminology_applied': len(terminology_map),
            'quality_score': quality_score,
            'terminology_mapping': terminology_map
        }
    
    def _extract_terminology(self, text: str, language: str, 
                            domain: str = None) -> List[Dict]:
        """提取需要特殊处理的术语"""
        nlp = self.nlp_models.get(language)
        if not nlp:
            return []
        
        doc = nlp(text)
        terminology = []
        
        for token in doc:
            # 检查是否为重要术语
            if token.pos_ in ['NOUN', 'PROPN'] and len(token.text) > 3:
                in_terminology_db = token.text.lower() in self.terminology_db
                
                if in_terminology_db or self._is_academic_term(token.text, domain):
                    terminology.append({
                        'original': token.text,
                        'position': token.idx,
                        'is_domain_specific': in_terminology_db or domain is not None
                    })
        
        return terminology

class TranslationQualityEvaluator:
    def evaluate(self, original_text: str, translated_text: str,
                source_lang: str, target_lang: str) -> Dict:
        """评估翻译质量的多个维度"""
        
        # 1. BLEU得分
        bleu_score = self._calculate_bleu(original_text, translated_text)
        
        # 2. 流畅性评分 (使用LLM)
        fluency_score = self._evaluate_fluency(translated_text, target_lang)
        
        # 3. 准确性评分
        accuracy_score = self._evaluate_accuracy(original_text, translated_text)
        
        # 4. 一致性检查
        consistency_score = self._check_consistency(original_text, translated_text)
        
        # 综合评分
        overall_score = (
            bleu_score * 0.25 +
            fluency_score * 0.35 +
            accuracy_score * 0.25 +
            consistency_score * 0.15
        )
        
        return {
            'overall_score': overall_score,
            'bleu_score': bleu_score,
            'fluency_score': fluency_score,
            'accuracy_score': accuracy_score,
            'consistency_score': consistency_score,
            'quality_level': self._score_to_level(overall_score)
        }
    
    def _score_to_level(self, score: float) -> str:
        if score >= 0.9:
            return "Excellent"
        elif score >= 0.75:
            return "Good"
        elif score >= 0.6:
            return "Fair"

            'slide_count': len(slide_structure),
            'template_used': template,
            'color_scheme': color_scheme,
            'slides_data': slides_data,
            'speaker_notes': speaker_notes,
            'estimated_duration_minutes': len(slide_structure) * 2
        }
    
    def _plan_slide_structure(self, content: Dict, template_config: Dict) -> List[Dict]:
        """规划幻灯片结构"""
        slides = []
        
        # 标题页
        slides.append({
            'type': 'title',
            'title': content['title'],
            'subtitle': content.get('subtitle', ''),
            'author': content.get('author', '')
        })
        
        # 目录页
        sections = content['sections']
        slides.append({
            'type': 'toc',
            'title': 'Outline',
            'items': [s['title'] for s in sections]
        })
        
        # 内容页
        max_points = template_config['max_points_per_slide']
        for section in sections:
            # 章节标题页
            slides.append({
                'type': 'section_title',
                'title': section['title'],
                'description': section.get('description', '')
            })
            
            # 内容幻灯片
            points = section['content_points']
            for i in range(0, len(points), max_points):
                slide_points = points[i:i+max_points]
                
                # 判断是否需要图表
                has_data = any(p.get('data') for p in slide_points)
                
                slides.append({
                    'type': 'content_with_visualization' if has_data else 'content',
                    'title': section['title'],
                    'content_points': slide_points,
                    'visualization': self._plan_visualization(slide_points) if has_data else None
                })
        
        # 结论页
        slides.append({
            'type': 'conclusion',
            'title': 'Conclusion',
            'key_takeaways': content.get('key_takeaways', [])
        })
        
        # 谢谢页
        slides.append({
            'type': 'thank_you',
            'title': 'Thank You',
            'contact_info': content.get('contact_info', {})
        })
        
        return slides
    
    def _create_slide(self, prs: Presentation, slide_num: int, 
                     slide_info: Dict, color_scheme: Dict, 
                     template: str) -> Any:
        """创建单个幻灯片"""
        
        slide_layout = self._get_slide_layout(prs, slide_info['type'])
        slide = prs.slides.add_slide(slide_layout)
        
        # 应用配色方案
        background = slide.background
        fill = background.fill
        fill.solid()
        fill.fore_color.rgb = RGBColor(*color_scheme['background'])
        
        # 根据幻灯片类型填充内容
        if slide_info['type'] == 'title':
            self._fill_title_slide(slide, slide_info, color_scheme)
        elif slide_info['type'] == 'content':
            self._fill_content_slide(slide, slide_info, color_scheme)
        elif slide_info['type'] == 'content_with_visualization':
            self._fill_content_with_viz_slide(slide, slide_info, color_scheme)
        
        return slide
    
    def _fill_content_slide(self, slide, slide_info: Dict, color_scheme: Dict):
        """填充内容幻灯片"""
        # 标题
        title_box = slide.shapes.title
        title_box.text = slide_info['title']
        title_frame = title_box.text_frame
        title_frame.paragraphs[0].font.size = Pt(44)
        title_frame.paragraphs[0].font.bold = True
        title_frame.paragraphs[0].font.color.rgb = RGBColor(*color_scheme['primary'])
        
        # 内容要点
        if 'content_points' in slide_info:
            left = Inches(0.5)
            top = Inches(1.5)
            width = Inches(9)
            height = Inches(5.5)
            
            text_box = slide.shapes.add_textbox(left, top, width, height)
            text_frame = text_box.text_frame
            text_frame.word_wrap = True
            
            for point in slide_info['content_points']:
                p = text_frame.add_paragraph()
                p.text = point['text']
                p.level = point.get('level', 0)
                p.font.size = Pt(24 - p.level * 4)
                p.font.color.rgb = RGBColor(*color_scheme['text'])
                p.space_before = Pt(6)
                p.space_after = Pt(6)

    async def _generate_speaker_notes(self, slide_structure: List[Dict]) -> Dict:
        """为每张幻灯片生成演讲稿"""
        notes = {}
        
        for idx, slide_info in enumerate(slide_structure):
            if slide_info['type'] in ['title', 'thank_you']:
                continue
            
            prompt = f"""
            为以下幻灯片生成简洁的演讲稿（100-150字）：
            
            标题: {slide_info.get('title', '')}
            内容: {str(slide_info.get('content_points', []))}
            
            演讲稿应该：
            - 总结要点
            - 提供上下文
            - 建议表达方式
            """
            
            speaker_note = await self.llm.generate(prompt)
            notes[f"slide_{idx}"] = speaker_note
        
        return notes

```python
# 实习周报生成引擎
from datetime import datetime, timedelta
from collections import defaultdict

class InternshipReportGenerator:
    def __init__(self):
        self.summarizer = pipeline("summarization", model="facebook/bart-large-cnn")
        self.classifier = self._load_task_classifier()
        self.report_templates = self._load_templates()
    
    TASK_CATEGORIES = {
        'development': '开发工作',
        'testing': '测试工作',
        'documentation': '文档编写',
        'meeting': '会议参与',
        'learning': '学习培训',
        'communication': '沟通协调',
        'other': '其他工作'
    }
    
    async def generate_weekly_report(self, user_id: str, 
                                    week_start: datetime) -> Dict:
        """生成周报"""
        week_end = week_start + timedelta(days=6)
        
        # 1. 获取这周的所有工作日志
        work_logs = await self._fetch_work_logs(user_id, week_start, week_end)
        
        # 2. 分析工作内容
        work_analysis = self._analyze_work_content(work_logs)
        
        # 3. 提取关键信息
        key_achievements = self._extract_achievements(work_analysis)
        challenges = self._identify_challenges(work_logs)
        learning_points = self._extract_learning_points(work_logs)
        
        # 4. 生成周报内容
        report_content = {
            'week_start': week_start.strftime('%Y-%m-%d'),
            'week_end': week_end.strftime('%Y-%m-%d'),
            'summary': self._generate_summary(work_analysis),
            'work_summary': {
                'total_hours': work_analysis['total_hours'],
                'tasks_completed': work_analysis['task_count'],
                'by_category': work_analysis['by_category']
            },
            'achievements': key_achievements,
            'challenges': challenges,
            'learning': learning_points,
            'next_week_plan': self._generate_next_week_plan(work_analysis),
            'metrics': self._calculate_metrics(work_analysis)
        }
        
        # 5. 应用模板生成最终文档
        formatted_report = self._format_report(report_content)
        
        return {
            'report_content': formatted_report,
            'statistics': work_analysis,
            'export_formats': ['docx', 'pdf', 'markdown']
        }
    
    def _analyze_work_content(self, work_logs: List[Dict]) -> Dict:
        """分析工作内容"""
        analysis = {
            'total_hours': 0,
            'task_count': 0,
            'by_category': defaultdict(list),
            'daily_breakdown': defaultdict(float),
            'key_projects': []
        }
        
        for log in work_logs:
            # 提取工作时间
            hours = log.get('hours_spent', 0)
            analysis['total_hours'] += hours
            
            # 分类
            category = self._classify_task(log['description'])
            analysis['by_category'][category].append(log)
            
            # 每日统计
            date = log['date'].strftime('%Y-%m-%d')
            analysis['daily_breakdown'][date] += hours
            
            analysis['task_count'] += 1
        
        # 项目统计
        project_counts = {}
        for log in work_logs:
            project = log.get('project', 'Other')
            project_counts[project] = project_counts.get(project, 0) + 1
        
        analysis['key_projects'] = sorted(
            project_counts.items(), 
            key=lambda x: x[1], 
            reverse=True
        )[:3]
        
        return analysis
    
    def _extract_achievements(self, work_analysis: Dict) -> List[str]:
        """提取主要成果"""
        achievements = []
        
        # 按类别总结
        for category, logs in work_analysis['by_category'].items():
            if logs:
                # 为该类别生成摘要
                tasks_text = ' '.join([log['description'] for log in logs[:5]])
                
                summary = self.summarizer(
                    tasks_text, 
                    max_length=50, 
                    min_length=20,
                    do_sample=False
                )
                
                if summary:
                    achievements.append({
                        'category': self.TASK_CATEGORIES.get(category, category),
                        'achievement': summary[0]['summary_text'],
                        'count': len(logs)
                    })
        
        return achievements
    
    def _format_report(self, report_content: Dict) -> str:
        """格式化周报"""
        template = self.report_templates['default']
        
        formatted = f"""
# {report_content['week_start']} - {report_content['week_end']} 周报

## 本周概况
{report_content['summary']}

## 工作总结
- 总工作时长: {report_content['work_summary']['total_hours']} 小时
- 完成任务数: {report_content['work_summary']['tasks_completed']} 项

### 按类别统计
"""
        
        for category, logs in report_content['work_summary']['by_category'].items():
            formatted += f"\n- {category}: {len(logs)} 项"
        
        formatted += "\n\n## 主要成果\n"
        for achievement in report_content['achievements']:
            formatted += f"\n### {achievement['category']}\n"
            formatted += f"{achievement['achievement']}\n"
        
        formatted += "\n## 遇到的问题与解决方案\n"
        for challenge in report_content['challenges']:
            formatted += f"- {challenge}\n"
        
        formatted += "\n## 学习与收获\n"
        for point in report_content['learning']:
            formatted += f"- {point}\n"
        
        formatted += f"\n## 下周计划\n"
        formatted += report_content['next_week_plan']
        
        return formatted
```
